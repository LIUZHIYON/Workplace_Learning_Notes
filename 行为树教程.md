# ROS2 行为树实战指南 — 提醒 + WebSocket 节点

## 什么是行为树？

行为树（Behavior Tree, BT）就是用**树形结构**来控制机器人做什么。对比传统状态机，BT 更灵活、可复用、可热插拔。

### 核心节点类型

```
           ┌──────────┐
           │  Root     │  — 树的入口
           └────┬─────┘
                │
         ┌──────┴──────┐
         │  Sequence    │  — 顺序执行：A→B→C，有一个失败就停
         └──────┬───────┘
         ┌──┼──┐  ┌──┼──┐  ┌──┼──┐
         │  A  │  │  B  │  │  C  │
         └─────┘  └─────┘  └─────┘
```

| 节点类型 | 行为 | 类比 |
|---------|------|------|
| **Sequence** (→) | 从左到右依次执行，全成功才成功 | "且" / 串行 |
| **Fallback/Selector** (?) | 从左到右尝试，第一个成功的就返回成功 | "或" / 容错 |
| **Condition** | 检查条件，成功/失败（不改状态） | 如果... |
| **Action** | 执行动作，成功/失败/运行中 | 做... |
| **Decorator** | 修饰子节点（取反/重试/超时） | 修饰器 |
| **ReactiveSequence** | Sequence + 每 tick 重新检查条件 | 高响应版 Sequence |

## 轻量 Python 行为树引擎

下面是一个独立引擎，**零外部依赖**，直接在你的 ROS2 项目里用。

### 1. 行为树引擎代码

```python
# bt_core.py — 行为树核心引擎
from enum import Enum
from typing import List, Optional, Callable


class Status(Enum):
    SUCCESS = "success"
    FAILURE = "failure"
    RUNNING = "running"


class Node:
    """行为树节点基类"""
    def __init__(self, name: str):
        self.name = name
        self._children: List[Node] = []

    def add_child(self, child: Node) -> 'Node':
        self._children.append(child)
        return self  # 支持链式调用

    def tick(self) -> Status:
        raise NotImplementedError


class Action(Node):
    """动作节点 — 执行具体操作"""
    def __init__(self, name: str, fn: Callable[[], Status]):
        super().__init__(name)
        self._fn = fn

    def tick(self) -> Status:
        return self._fn()


class Condition(Node):
    """条件节点 — 检查条件是否满足"""
    def __init__(self, name: str, fn: Callable[[], bool]):
        super().__init__(name)
        self._fn = fn

    def tick(self) -> Status:
        return Status.SUCCESS if self._fn() else Status.FAILURE


class Sequence(Node):
    """Sequence: 全部成功才成功，一失败就停"""
    def __init__(self, name: str, children: Optional[List[Node]] = None):
        super().__init__(name)
        if children:
            self._children = children

    def tick(self) -> Status:
        for child in self._children:
            status = child.tick()
            if status == Status.FAILURE:
                return Status.FAILURE
            if status == Status.RUNNING:
                return Status.RUNNING
        return Status.SUCCESS


class Fallback(Node):
    """Fallback/Selector: 有一个成功就成功，一成功就停"""
    def __init__(self, name: str, children: Optional[List[Node]] = None):
        super().__init__(name)
        if children:
            self._children = children

    def tick(self) -> Status:
        for child in self._children:
            status = child.tick()
            if status == Status.SUCCESS:
                return Status.SUCCESS
            if status == Status.RUNNING:
                return Status.RUNNING
        return Status.FAILURE


class ReactiveSequence(Node):
    """ReactiveSequence: 每 tick 重新检查条件（有记忆版本）"""
    def __init__(self, name: str, children: Optional[List[Node]] = None):
        super().__init__(name)
        if children:
            self._children = children
        self._running_child = 0

    def tick(self) -> Status:
        # 每次 tick 都重新检查所有条件节点（前面的节点）
        # 但已 RUNNING 的子节点直接继续
        for i, child in enumerate(self._children):
            if i < self._running_child:
                # 如果之前跑过的条件现在失败了，从头开始
                if isinstance(child, Condition):
                    if child.tick() == Status.FAILURE:
                        self._running_child = 0
                        return Status.FAILURE
                continue

            status = child.tick()
            if status == Status.RUNNING:
                self._running_child = i
                return Status.RUNNING
            if status == Status.FAILURE:
                self._running_child = 0
                return Status.FAILURE

        self._running_child = 0
        return Status.SUCCESS


class Retry(Node):
    """Retry 装饰器 — 失败重试指定次数"""
    def __init__(self, name: str, child: Node, max_attempts: int = 3):
        super().__init__(name)
        self._child = child
        self._max_attempts = max_attempts
        self._attempts = 0

    def tick(self) -> Status:
        while self._attempts < self._max_attempts:
            status = self._child.tick()
            if status == Status.SUCCESS:
                self._attempts = 0
                return Status.SUCCESS
            if status == Status.RUNNING:
                return Status.RUNNING
            self._attempts += 1
        self._attempts = 0
        return Status.FAILURE


class Inverter(Node):
    """取反装饰器"""
    def __init__(self, name: str, child: Node):
        super().__init__(name)
        self._child = child

    def tick(self) -> Status:
        status = self._child.tick()
        if status == Status.SUCCESS:
            return Status.FAILURE
        if status == Status.FAILURE:
            return Status.SUCCESS
        return Status.RUNNING


class BehaviorTree:
    """行为树 — 提供 tick() 入口"""
    def __init__(self, root: Node, name: str = "root"):
        self.root = root
        self.name = name

    def tick(self) -> Status:
        return self.root.tick()
```

### 3. 完整示例：提醒 + WebSocket 联动

```python
"""
reminder_bt.py — 提醒功能行为树
依赖: bt_core.py
ROS2 环境: Humble
"""

import rclpy
from rclpy.node import Node as ROSNode
import requests
import json

# 导入上面的行为树引擎
from bt_core import (
    BehaviorTree, Status, Sequence, Fallback,
    ReactiveSequence, Condition, Action, Retry, Inverter
)


class ReminderBTNode(ROSNode):
    """行为树 + ROS2 节点"""

    def __init__(self):
        super().__init__('reminder_bt_node')

        # 配置参数
        self.declare_parameter('reminder_api_url', 'http://localhost:5000')
        self.declare_parameter('check_interval_ms', 100)  # BT tick 间隔
        self.declare_parameter('tts_topic', '/tts/text')

        # 状态变量
        self.pending_reminders = []
        self.current_reminder = None
        self.tts_ready = False

        # 构建行为树
        self.bt = self._build_behavior_tree()

        # 定时器 — 每 100ms tick 一次
        interval = self.get_parameter('check_interval_ms').value / 1000.0
        self.create_timer(interval, self._tick)

        self.get_logger().info('行为树节点已启动')

    # ────────────── 条件节点回调 ──────────────

    def _has_pending_reminder(self) -> bool:
        """Condition: 检查是否有待触发的提醒"""
        try:
            url = f"{self.get_parameter('reminder_api_url').value}/api/stats"
            resp = requests.get(url, timeout=3)
            data = resp.json()
            if data.get('success'):
                pending = data['data'].get('pending', 0)
                if pending > 0:
                    # 获取具体内容
                    self.pending_reminders = self._fetch_pending()
                    return len(self.pending_reminders) > 0
            return False
        except Exception as e:
            self.get_logger().warn(f'检查提醒失败: {e}')
            return False

    def _is_tts_busy(self) -> bool:
        """Condition: TTS 是否空闲"""
        return not self.tts_ready

    # ────────────── 动作节点回调 ──────────────

    def _fetch_pending(self):
        """Action: 获取待处理提醒列表"""
        try:
            url = f"{self.get_parameter('reminder_api_url').value}/api/reminders?status=pending"
            resp = requests.get(url, timeout=3)
            data = resp.json()
            return data.get('data', [])
        except:
            return []

    def _send_tts(self) -> Status:
        """Action: 通过 ROS2 话题发送 TTS"""
        if not self.current_reminder:
            return Status.FAILURE

        text = self.current_reminder.get('content', '')
        if not text:
            return Status.FAILURE

        # 通过话题发送给 doubao_tts_node
        msg = std_msgs.msg.String()
        msg.data = text
        self.tts_pub.publish(msg)
        self.get_logger().info(f'TTS 发送: {text}')
        return Status.SUCCESS

    def _send_websocket_notify(self) -> Status:
        """Action: 通过 WebSocket 发送通知"""
        if not self.current_reminder:
            return Status.FAILURE

        try:
            # 通过 ROS2 话题发送到 websocket 节点
            msg = std_msgs.msg.String()
            msg.data = json.dumps({
                'type': 'reminder_triggered',
                'content': self.current_reminder.get('content', ''),
                'time': self.current_reminder.get('reminder_time', ''),
                'id': self.current_reminder.get('id', 0),
            })
            self.ws_pub.publish(msg)
            return Status.SUCCESS
        except Exception as e:
            self.get_logger().error(f'WebSocket 通知失败: {e}')
            return Status.FAILURE

    def _mark_triggered(self) -> Status:
        """Action: 标记提醒已触发"""
        if not self.current_reminder:
            return Status.FAILURE
        rid = self.current_reminder.get('id')
        # 调用提醒 API 标记触发
        try:
            url = f"{self.get_parameter('reminder_api_url').value}/api/reminders/{rid}/trigger"
            requests.post(url, timeout=3)
            self.current_reminder = None
            return Status.SUCCESS
        except:
            return Status.FAILURE

    def _take_first_reminder(self) -> Status:
        """Action: 从队列取第一个提醒"""
        if self.pending_reminders:
            self.current_reminder = self.pending_reminders.pop(0)
            return Status.SUCCESS
        return Status.FAILURE

    # ────────────── 构建行为树 ──────────────

    def _build_behavior_tree(self) -> BehaviorTree:
        """
        构建的行为树结构:

        ReactiveSequence("提醒主循环")
        ├─ Condition("有待触发提醒?")
        ├─ Sequence("处理提醒")
        │  ├─ Action("取第一条")
        │  ├─ ReactiveSequence("播报")
        │  │  ├─ Condition("TTS空闲?")
        │  │  ├─ Action("发送TTS")
        │  │  ├─ Retry("WebSocket通知", max=3)
        │  │  │  └─ Action("发WS通知")
        │  │  └─ Action("标记已触发")
        │  └─ Action("记录日志")
        └─ Action("等待") ← Fallback 可选
        """
        root = ReactiveSequence("提醒主循环", [
            Condition("有待触发提醒?", self._has_pending_reminder),
            Sequence("处理提醒", [
                Action("取第一条", lambda: self._take_first_reminder()),
                ReactiveSequence("播报提醒", [
                    Condition("TTS空闲?", lambda: self.tts_ready),
                    Action("发送TTS", self._send_tts),
                    Retry("WebSocket重试", Action("发WS通知", self._send_websocket_notify), max_attempts=3),
                    Action("标记已触发", self._mark_triggered),
                ]),
                Action("记录日志", lambda: self._log_complete()),
            ]),
        ])

        return BehaviorTree(root, name="reminder_bt")

    def _log_complete(self) -> Status:
        self.get_logger().info(f'提醒播报完成')
        return Status.SUCCESS

    # ────────────── 主循环 ──────────────

    def _tick(self):
        """每 100ms tick 行为树"""
        status = self.bt.tick()
        if status == Status.RUNNING:
            pass  # 动作进行中
        elif status == Status.SUCCESS:
            pass  # 无待办
        # FAILURE 自动由 Fallback/Sequence 处理


def main():
    rclpy.init()
    node = ReminderBTNode()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```

# bt_visualize.py
def print_tree(node, indent=""):
    """打印行为树结构"""
    status_map = {
        Status.SUCCESS: "✅",
        Status.FAILURE: "❌",
        Status.RUNNING: "⏳",
    }
    icon = status_map.get(type(node).__name__, "●")
    print(f"{indent}{icon} {type(node).__name__}({node.name})")
    if hasattr(node, '_children'):
        for child in node._children:
            print_tree(child, indent + "  ├─ ")

## 你的场景适用性

| 场景 | BT 是否合适 | 原因 |
|------|:-----------:|------|
| 定时检查+播报提醒 | ✅ | 条件+动作天然匹配 |
| WebSocket 通知容错 | ✅ | Retry 装饰器搞定断线重发 |
| 多个提醒并发 | ✅ | Sequence 依次处理 |
| 打断/优先级 | ✅ | Fallback 实现高优插播 |
| 纯顺序脚本 | ❌ 用 if-else | 简单逻辑不需要 BT |

**行为树的优势在混搭场景才明显**：比如"有高优消息时打断当前播报，播完再续播"——用 Fallback + ReactiveSequence 几行搞定，用 if-else 会写得很难看。
